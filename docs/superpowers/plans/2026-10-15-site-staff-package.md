# Site Staff Package Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a portable Windows folder and zip that start the site staff app with a double-click, carrying their own Python, the locked dependencies with processor-only PyTorch, the newest model, and the app.

**Architecture:** A build command installs a standalone CPython with uv, syncs the locked `model-cpu` dependency set into it, removes build-only files, compiles bytecode, copies the app and model, writes a batch launcher and plain instructions, runs a self-check with the package's own Python, and zips the folder.
The batch launcher runs `frog_classifier.app.launch`, which checks the computer, reuses a running app or starts Streamlit for this computer only, waits until it answers, and opens the browser.
Results live in `%LOCALAPPDATA%\Frog Call Scanner`; the model stays in the package and is found through `FROG_CLASSIFIER_MODELS`.

**Tech Stack:** Python 3.13, uv 0.12 (`uv python install`, `uv sync`), Streamlit 1.54, PyTorch 2.14.1 CPU build, Windows batch, `zipfile`, `unittest`.

**Spec:** `docs/superpowers/specs/2026-10-15-site-staff-package-design.md`

## Global Constraints

- The package runs on Windows 10 or 11, 64-bit; the build runs on Windows with uv.
- Packaged Python: CPython 3.13.14 from `uv python install`, installed with `--no-bin --no-registry` so the build computer's own Python setup is untouched.
- Dependencies come only from `uv.lock`, through `uv sync --frozen --no-default-groups --group model-cpu --no-editable`.
- The `model` group stays the default; development environments keep PyTorch `+cu130`, and `requirements_labeler.txt` does not change.
- Preferred port 8571; Streamlit flags `--server.address localhost --server.headless true --browser.gatherUsageStats false --client.toolbarMode minimal --server.fileWatcherType none`.
- Data folder `%LOCALAPPDATA%\Frog Call Scanner`; model folder variable `FROG_CLASSIFIER_MODELS`; data root variable `FROG_CLASSIFIER_ROOT`.
- Every path inside the package folder is at most 160 characters.
- `msvcp140.dll` missing means: install Microsoft's Visual C++ Redistributable (x64) from `https://aka.ms/vs/17/release/vc_redist.x64.exe`.
- `frog_classifier.app` modules never import PyTorch.
- No em dash anywhere; one sentence per physical line in long Markdown.
- `dist/` is generated and ignored by Git; nothing under it is committed.
- Commit messages carry no co-author trailer.

## Review Focus

- Started from inside the zip without extracting it: the launcher says to extract the whole download first, instead of a cryptic "not recognized" error (Task 4 test).
- A proxy configured for the internet on a work computer: the launcher's local health check still reaches the app (Task 3 test).
- Python variables set by other software, such as `PYTHONHOME` from GIS tools: the launcher, Streamlit, and every scan ignore them (Task 3 tests).
- A first start slowed by antivirus scanning, or Streamlit failing to start: the launcher waits, then says what happened instead of hanging (Task 3 tests).
- A build from changed or new source files: refused unless marked uncommitted, so every package maps to a commit (Task 6 tests).

@dataclass(frozen=True)
class AppPaths:
    root: Path
    # The site staff package keeps its model beside the app and its data elsewhere.
    model_folder: Path | None = None

    @property
    def models(self) -> Path:
        return self.model_folder if self.model_folder is not None else self.root / "models"

    @property
    def uploads(self) -> Path:
        return self.root / "raw" / "uploads"

    @property
    def detections(self) -> Path:
        return self.root / "results" / "detections"

    root = os.environ.get(ROOT_VARIABLE)

    return AppPaths(Path(root) if root else default_root, Path(models) if models else None)

    def do_GET(self) -> None:
        self.send_response(200 if self.path == "/_stcore/health" else 404)

    """A stand-in for a running app: it answers the health address only."""
    server = http.server.HTTPServer(("127.0.0.1", 0), _Health)
    threading.Thread(target=server.serve_forever, daemon=True).start()
    test.addCleanup(server.server_close)

"""Start the frog call scanner for a person: the browser opens, and closing the window stops it.

The site staff package's Start Frog Call Scanner.bat runs
python -I -m frog_classifier.app.launch --app APP --models MODELS --data DATA.
Streamlit, and every scan it starts, shares this console window, so closing the
window ends them all, and the app continues an interrupted scan on request.
"""

from __future__ import annotations

import argparse
import ctypes
import os
import socket
import subprocess
import sys
import time
import urllib.request
import webbrowser
from collections.abc import Mapping
from pathlib import Path
from typing import IO

from .paths import MODELS_VARIABLE, ROOT_VARIABLE


PREFERRED_PORT = 8571
READY_TIMEOUT_S = 120.0
RUNTIME_ADDRESS = "https://aka.ms/vs/17/release/vc_redist.x64.exe"
# Headless mode skips Streamlit's first-run email question, which would wait for
# typing in the window; the launcher opens the browser itself once the app answers.
STREAMLIT_SETTINGS = (
    "--server.address", "localhost",
    "--server.headless", "true",
    "--browser.gatherUsageStats", "false",
    "--client.toolbarMode", "minimal",
    "--server.fileWatcherType", "none",
)
_POLL_SECONDS = 0.25
# Requests to the app on this computer must never go through a proxy set up for the internet.
_DIRECT = urllib.request.build_opener(urllib.request.ProxyHandler({}))


class LaunchError(RuntimeError):
    pass


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Start the frog call scanner and open it in the browser.")
    parser.add_argument("--app", type=Path, required=True, help="The app page, classifier_app/app.py.")
    parser.add_argument("--models", type=Path, required=True, help="Folder of saved model runs.")
    parser.add_argument("--data", type=Path, required=True, help="Folder for results and uploaded recordings.")
    parser.add_argument("--port", type=int, default=PREFERRED_PORT, help="Preferred port. Default: %(default)s.")
    parser.add_argument("--no-browser", action="store_true", help="Do not open the browser.")
    args = parser.parse_args(argv)
    try:
        return _launch(args.app, args.models, args.data, args.port, open_browser=not args.no_browser)
    except LaunchError as error:
        print(f"\nThe frog call scanner cannot start: {error}.\n")
        return 1
    except KeyboardInterrupt:
        return 0


def _launch(app_file: Path, models: Path, data: Path, preferred_port: int, *, open_browser: bool) -> int:
    if not app_file.is_file() or not models.is_dir():
        missing = app_file if not app_file.is_file() else models
        raise LaunchError(f"{missing} is missing; unzip the whole package again")
    prepare_data_folder(data)
    problem = runtime_problem()
    if problem is not None:
        raise LaunchError(problem)
    port, running = choose_port(preferred_port)
    url = f"http://localhost:{port}"
    if running:
        print(f"The frog call scanner is already running; opening {url}")
        _show(url, open_browser)
        return 0
    print("Starting the frog call scanner...")
    process = start_app(app_file, port, child_environment(os.environ, data, models))
    try:
        wait_until_ready(port, process)
        print(f"\nThe frog call scanner is running at {url}")
        print("Keep this window open while you use it. Closing it stops the scanner and any scan.\n")
        _show(url, open_browser)
        return process.wait()
    finally:
        if process.poll() is None:
            process.terminate()


def prepare_data_folder(data: Path) -> Path:
    if not data.is_absolute():
        raise LaunchError(f"the data folder {data} is not a full path")
    try:
        data.mkdir(parents=True, exist_ok=True)
    except OSError as error:
        raise LaunchError(f"the data folder {data} cannot be created: {error}") from error
    return data


def runtime_problem() -> str | None:
    """Why the detector cannot load on this computer, or None.

    PyTorch needs msvcp140.dll, which neither Python nor PyTorch ships.
    """
    if sys.platform != "win32":
        return None
    try:
        ctypes.WinDLL("msvcp140.dll")
    except OSError:
        return (
            "this computer lacks Microsoft's Visual C++ Redistributable (x64), which the detector needs; "
            f"ask whoever looks after this computer to install it from {RUNTIME_ADDRESS}, then start the scanner again"
        )
    return None


def app_answers(port: int) -> bool:
    try:
        with _DIRECT.open(f"http://127.0.0.1:{port}/_stcore/health", timeout=2) as response:
            return response.status == 200
    except (OSError, ValueError):
        return False


def port_is_free(port: int) -> bool:
    with socket.socket() as probe:
        try:
            probe.bind(("127.0.0.1", port))
        except OSError:
            return False
    return True


def free_port() -> int:
    with socket.socket() as probe:
        probe.bind(("127.0.0.1", 0))
        return probe.getsockname()[1]


def choose_port(preferred: int) -> tuple[int, bool]:
    """The port to use, and whether the app already runs there."""
    if app_answers(preferred):
        return preferred, True
    if port_is_free(preferred):
        return preferred, False
    return free_port(), False


def child_environment(environ: Mapping[str, str], data: Path, models: Path) -> dict[str, str]:
    """The person's environment without their Python settings, pointing the app at its folders.

    GIS and other tools sometimes set PYTHONHOME or PYTHONPATH for their own Python;
    scans inherit this environment, so those settings must not reach them.
    """
    environment = {name: value for name, value in environ.items() if not name.upper().startswith("PYTHON")}
    environment["PYTHONNOUSERSITE"] = "1"
    environment[ROOT_VARIABLE] = str(data)
    environment[MODELS_VARIABLE] = str(models)
    return environment


def streamlit_command(app_file: Path, port: int) -> list[str]:
    return [sys.executable, "-I", "-m", "streamlit", "run", str(app_file), "--server.port", str(port), *STREAMLIT_SETTINGS]


def start_app(
    app_file: Path,
    port: int,
    environment: Mapping[str, str],
    *,
    log: IO[bytes] | None = None,
) -> subprocess.Popen:
    """Start Streamlit; its messages go to this window, or to log when given."""
    return subprocess.Popen(
        streamlit_command(app_file, port),
        env=dict(environment),
        stdin=subprocess.DEVNULL,
        stdout=log,
        stderr=subprocess.STDOUT if log is not None else None,
    )


def wait_until_ready(port: int, process: subprocess.Popen, *, timeout_s: float = READY_TIMEOUT_S) -> None:
    """Wait for the app to answer; a first start can be slow while Windows checks the new files."""
    deadline = time.monotonic() + timeout_s
    while not app_answers(port):
        if process.poll() is not None:
            raise LaunchError(
                f"the app stopped while starting, with exit code {process.returncode}; the messages above say why"
            )
        if time.monotonic() > deadline:
            raise LaunchError(f"the app did not answer within {timeout_s:.0f} seconds")
        time.sleep(_POLL_SECONDS)


def _show(url: str, open_browser: bool) -> None:
    if open_browser and not webbrowser.open(url):
        print(f"Open {url} in your web browser.")


if __name__ == "__main__":
    raise SystemExit(main())

def prune(python_dir: Path) -> None:
    """Remove the files only compiling needs from an installed Python."""
    for folder in PRUNED_FOLDERS:
        if (python_dir / folder).is_dir():
            shutil.rmtree(python_dir / folder)
    for path in python_dir.rglob(f"*{PRUNED_SUFFIX}"):
        if path.is_file():
            path.unlink()


def paths_over_budget(package_dir: Path, budget: int = PATH_BUDGET) -> list[str]:
    """Paths inside package_dir longer than budget characters, as Windows writes them, longest first."""
    over = (path for path in _relative_paths(package_dir) if len(path) > budget)
    return sorted(over, key=len, reverse=True)


def longest_path(package_dir: Path) -> int:
    return max((len(path) for path in _relative_paths(package_dir)), default=0)


def _relative_paths(package_dir: Path) -> list[str]:
    return [str(PureWindowsPath(path.relative_to(package_dir))) for path in package_dir.rglob("*")]

"""Check that the site staff package runs on its own.

The build runs python -I -m frog_classifier.packaging.selfcheck PACKAGE with the
package's own Python and an environment holding only what Windows programs need,
so nothing from the build computer can help a check pass. PyTorch is imported
only inside the checks that need it.
"""

from __future__ import annotations

import argparse
import os
import subprocess
import sys
import tempfile
import wave
from collections.abc import Callable, Iterator, Mapping, Sequence
from contextlib import contextmanager
from pathlib import Path

import numpy as np

from frog_classifier.app import launch
from frog_classifier.app.models import newest_model
from frog_classifier.app.paths import MODELS_VARIABLE, ROOT_VARIABLE

from .contents import APP_FILE, MODELS_FOLDER


class SelfCheckError(RuntimeError):
    pass


def check_isolated(package: Path, search_path: Sequence[str]) -> None:
    """Every import location is inside the package."""
    inside = package.resolve()
    for entry in search_path:
        location = Path(entry).resolve()
        if not location.is_relative_to(inside):
            raise SelfCheckError(f"Python imports from {location}, outside the package")


def check_app_without_pytorch() -> None:
    """The app's own process never loads PyTorch; only scans do."""
    program = (
        "import importlib, pkgutil, sys, frog_classifier.app as app\n"
        "for module in pkgutil.iter_modules(app.__path__):\n"
        "    importlib.import_module(f'frog_classifier.app.{module.name}')\n"
        "print('torch' in sys.modules)\n"
    )
    result = subprocess.run([sys.executable, "-I", "-c", program], capture_output=True, text=True)
    if result.returncode != 0:
        raise SelfCheckError(f"the app modules do not import: {result.stderr.strip()}")
    if result.stdout.strip() != "False":
        raise SelfCheckError("importing the app modules loads PyTorch")


def check_cpu_pytorch() -> None:
    """PyTorch is the processor build the package is meant to carry."""
    import torch

    if torch.version.cuda is not None:
        raise SelfCheckError(f"PyTorch {torch.__version__} is a graphics-card build, not the processor build")


def check_model(models: Path) -> Path:
    """The newest packaged model loads, which checks every file its run.json records."""
    from frog_classifier.training.saved import SavedRunError, load_saved_run

    run_dir = newest_model(models).run_dir
    try:
        load_saved_run(run_dir)
    except SavedRunError as error:
        raise SelfCheckError(str(error)) from error
    return run_dir


def check_scan(run_dir: Path) -> None:
    """A ten-second recording is scanned, with rendering worker processes, into two windows."""
    import torch

    from frog_classifier.inference.detect import detect_calls

    with tempfile.TemporaryDirectory() as temporary:
        folder = Path(temporary)
        _write_noise(folder / "recordings" / "20220123_010000.WAV", seconds=10)
        summary = detect_calls(
            run_dir, folder / "recordings", folder / "record", device=torch.device("cpu"), workers=2,
        )
    if (summary.processed, summary.failed, summary.windows) != (1, 0, 2):
        raise SelfCheckError(
            f"scanning a ten-second recording gave {summary.processed} processed, {summary.failed} failed, "
            f"and {summary.windows} windows, not 1, 0, and 2"
        )


def check_app_renders(app_file: Path, models: Path) -> None:
    """The app page runs in this Python and finds the packaged model."""
    from streamlit.testing.v1 import AppTest

    expected = f"Detector {newest_model(models).name}"
    with tempfile.TemporaryDirectory() as data, _environment({ROOT_VARIABLE: data, MODELS_VARIABLE: str(models)}):
        app = AppTest.from_file(str(app_file), default_timeout=60)
        app.run()
    if app.exception:
        raise SelfCheckError(f"the app page failed: {app.exception[0].value}")

def _environment(values: Mapping[str, str]) -> Iterator[None]:
    previous = {name: os.environ.get(name) for name in values}
    os.environ.update(values)
    try:
        yield
    finally:
        for name, value in previous.items():
            if value is None:
                os.environ.pop(name, None)
            else:
                os.environ[name] = value

from __future__ import annotations

import os
import shutil
import subprocess
import tempfile
import tomllib
import zipfile
from collections.abc import Callable, Mapping
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path, PurePosixPath

from .contents import (
    APP_FILE,
    BUILD_INFO_FILE,
    INSTRUCTIONS_FILE,
    LAUNCHER_FILE,
    MODELS_FOLDER,
    PACKAGE_FOLDER,
    PATH_BUDGET,
    PYTHON_FOLDER,
    BuildInfo,
    build_info_json,
    instructions_text,
    launcher_text,
    longest_path,
    paths_over_budget,
    prune,
    zip_name,
)


PYTHON_VERSION = "3.13.14"
DEPENDENCY_GROUP = "model-cpu"
# What ends up in the package; changes anywhere else do not make a build uncommitted.
PACKAGED_SOURCES = ("src", "classifier_app", "pyproject.toml", "uv.lock")
_ESSENTIAL_VARIABLES = (
    "SYSTEMROOT", "WINDIR", "COMSPEC", "PATHEXT", "SYSTEMDRIVE", "TEMP", "TMP",
    "USERPROFILE", "APPDATA", "LOCALAPPDATA", "HOMEDRIVE", "HOMEPATH", "USERNAME",
    "NUMBER_OF_PROCESSORS", "PROCESSOR_ARCHITECTURE", "OS",
)


class BuildError(RuntimeError):
    pass


@dataclass(frozen=True)
class BuiltPackage:
    folder: Path
    zip_path: Path
    info: BuildInfo
    files: int
    size_bytes: int
    longest_path: int


def find_uv(environ: Mapping[str, str]) -> str:
    """The uv that runs this build; uv run names itself in UV."""
    uv = environ.get("UV") or shutil.which("uv")
    if not uv:
        raise BuildError("uv was not found; run this command with uv run")
    return uv


def source_state(repo: Path) -> tuple[str, bool]:
    """The commit at HEAD, and whether the packaged sources differ from it."""
    commit = _git(repo, "rev-parse", "HEAD")
    changes = _git(repo, "status", "--porcelain", "--", *PACKAGED_SOURCES)
    return commit, bool(changes)


def python_install_command(uv: str, install_dir: Path) -> list[str]:
    # Without --no-bin and --no-registry, uv would add this Python to the build
    # computer's own tools folder and Windows registry.
    return [uv, "python", "install", PYTHON_VERSION, "--install-dir", str(install_dir), "--no-bin", "--no-registry"]


def sync_command(uv: str) -> list[str]:
    return [uv, "sync", "--frozen", "--no-default-groups", "--group", DEPENDENCY_GROUP, "--no-editable"]


def essential_environment(environ: Mapping[str, str]) -> dict[str, str]:
    """Only what Windows programs need, so the self-check cannot lean on the build computer's Python or uv."""
    environment = {name: environ[name] for name in _ESSENTIAL_VARIABLES if name in environ}
    system_root = environ.get("SYSTEMROOT", r"C:\Windows")
    environment["PATH"] = os.pathsep.join((rf"{system_root}\System32", system_root, rf"{system_root}\System32\Wbem"))
    return environment


def build_package(
    repo: Path,
    model_dir: Path,
    output: Path,
    *,
    uv: str,
    allow_uncommitted: bool,
    now: datetime,
    log: Callable[[str], None] = print,
) -> BuiltPackage:
    """Build, check, and zip the package; dist keeps it only when every check passes."""
    commit, uncommitted = source_state(repo)
    if uncommitted and not allow_uncommitted:
        raise BuildError(
            "the packaged sources have uncommitted changes; commit them, or pass --allow-uncommitted for a test build"
        )
    output.mkdir(parents=True, exist_ok=True)
    staging = Path(tempfile.mkdtemp(prefix=".building-", dir=output))
    try:
        package = staging / PACKAGE_FOLDER
        python_dir = package / PYTHON_FOLDER
        log(f"Installing Python {PYTHON_VERSION}...")
        _install_python(uv, staging / "python-download", python_dir)
        log("Installing the locked dependencies, with PyTorch for the processor...")
        _run(sync_command(uv), cwd=repo, env=_sync_environment(os.environ, python_dir))
        prune(python_dir)
        log("Compiling the Python modules...")
        _compile(python_dir)
        _copy_app(repo, model_dir, package)
        info = BuildInfo(
            version=_project_version(repo),
            commit=commit,
            uncommitted=uncommitted,
            built_at=now.strftime("%Y-%m-%dT%H:%M:%SZ"),
            python=PYTHON_VERSION,
            torch=_installed_version(python_dir, "torch"),
            model_run=model_dir.name,
        )
        (package / LAUNCHER_FILE).write_bytes(launcher_text().encode("ascii"))
        (package / INSTRUCTIONS_FILE).write_bytes(instructions_text(info).encode("utf-8"))
        (package / BUILD_INFO_FILE).write_bytes(build_info_json(info))
        too_long = paths_over_budget(package)
        if too_long:
            raise BuildError(
                f"{len(too_long)} paths inside the package are longer than {PATH_BUDGET} characters, "
                "the longest: " + "; ".join(too_long[:3])
            )
        log("Checking the package with its own Python...")
        _self_check(python_dir, package)
        files = [path for path in package.rglob("*") if path.is_file()]
        size_bytes = sum(path.stat().st_size for path in files)
        longest = longest_path(package)
        log("Zipping...")
        built_zip = staging / zip_name(info)
        _zip(package, built_zip)
        folder, zip_path = _keep(package, built_zip, output)
        return BuiltPackage(folder, zip_path, info, len(files), size_bytes, longest)
    finally:
        shutil.rmtree(staging, ignore_errors=True)


def _install_python(uv: str, install_dir: Path, python_dir: Path) -> None:
    _run(python_install_command(uv, install_dir))
    installed = [path for path in install_dir.glob(f"cpython-{PYTHON_VERSION}-*") if path.is_dir()]
    if len(installed) != 1:
        raise BuildError(f"expected one Python {PYTHON_VERSION} in {install_dir}, found {len(installed)}")
    python_dir.parent.mkdir(parents=True)
    shutil.move(installed[0], python_dir)


def _sync_environment(environ: Mapping[str, str], python_dir: Path) -> dict[str, str]:
    # uv run sets VIRTUAL_ENV to the development environment; the sync targets the package's Python.
    environment = {name: value for name, value in environ.items() if name != "VIRTUAL_ENV"}
    environment["UV_PROJECT_ENVIRONMENT"] = str(python_dir)
    return environment


def _compile(python_dir: Path) -> None:
    # Unchecked-hash bytecode is used whatever timestamps unzipping sets, so the first
    # start never recompiles. A few packages ship files that are not meant to compile,
    # such as templates, so compileall's failures are expected and harmless.
    subprocess.run(
        [
            str(python_dir / "python.exe"), "-I", "-m", "compileall", "-qq", "-f", "-j", "0",
            "--invalidation-mode", "unchecked-hash", str(python_dir / "Lib"),
        ],
        check=False,
    )


def _copy_app(repo: Path, model_dir: Path, package: Path) -> None:
    (package / APP_FILE).parent.mkdir(parents=True)
    shutil.copy2(repo / "classifier_app" / "app.py", package / APP_FILE)
    # The whole run, because loading it checks every file run.json records.
    shutil.copytree(model_dir, package / MODELS_FOLDER / model_dir.name)


def _project_version(repo: Path) -> str:
    return tomllib.loads((repo / "pyproject.toml").read_text(encoding="utf-8"))["project"]["version"]


def _installed_version(python_dir: Path, distribution: str) -> str:
    return _run([
        str(python_dir / "python.exe"), "-I", "-c",
        f"import importlib.metadata as metadata; print(metadata.version({distribution!r}))",
    ])


def _self_check(python_dir: Path, package: Path) -> None:
    result = subprocess.run(
        [str(python_dir / "python.exe"), "-I", "-m", "frog_classifier.packaging.selfcheck", str(package)],
        cwd=package,
        env=essential_environment(os.environ),
    )
    if result.returncode != 0:
        raise BuildError("the package failed its self-check; the lines above name each failed check")


def _zip(package: Path, zip_path: Path) -> None:
    with zipfile.ZipFile(zip_path, "w", compression=zipfile.ZIP_DEFLATED, compresslevel=6) as archive:
        for path in sorted(package.rglob("*")):
            if path.is_file():
                archive.write(path, PurePosixPath(PACKAGE_FOLDER, *path.relative_to(package).parts).as_posix())


def _keep(package: Path, built_zip: Path, output: Path) -> tuple[Path, Path]:
    folder = output / PACKAGE_FOLDER
    zip_path = output / built_zip.name
    try:
        if folder.exists():
            shutil.rmtree(folder)
        package.rename(folder)
        built_zip.replace(zip_path)
    except OSError as error:
        raise BuildError(
            f"the new package could not replace {folder}; close any scanner started from it and build again ({error})"
        ) from error
    return folder, zip_path


def _git(repo: Path, *arguments: str) -> str:
    result = subprocess.run(["git", *arguments], cwd=repo, capture_output=True, text=True)
    if result.returncode != 0:
        raise BuildError(f"git {' '.join(arguments)} failed: {result.stderr.strip()}")
    return result.stdout.strip()


def _run(command: list[str], *, cwd: Path | None = None, env: Mapping[str, str] | None = None) -> str:
    result = subprocess.run(command, cwd=cwd, env=None if env is None else dict(env), capture_output=True, text=True)
    if result.returncode != 0:
        raise BuildError(f"{' '.join(command[:3])} failed:\n{(result.stderr or result.stdout).strip()}")
    return result.stdout.strip()

"""Build the site staff package: a portable Windows folder and zip that start the app with a double-click.

The package carries its own Python, the locked dependencies with PyTorch for the
processor, one saved model, and the app. It goes to dist/ and is kept only when
the package's own Python passes the self-check. Building needs a committed tree,
so every package maps to one commit.
"""

from __future__ import annotations

import argparse
import os
import sys
from datetime import datetime, timezone
from pathlib import Path

from frog_classifier.app.models import ModelError, newest_model
from frog_classifier.packaging.build import BuildError, build_package, find_uv


def main() -> int:
    repo_root = Path(__file__).resolve().parents[1]
    parser = argparse.ArgumentParser(description=__doc__.split("\n\n", 1)[0])
    parser.add_argument("--model", type=Path, default=None, help="Saved run folder. Default: the newest run under models/.")
    parser.add_argument("--output", type=Path, default=Path("dist"), help="Output folder. Default: dist/.")
    parser.add_argument(
        "--allow-uncommitted",
        action="store_true",
        help="Build from uncommitted sources, for testing; the package is marked uncommitted.",
    )
    args = parser.parse_args()

    try:
        model_dir = _resolve(repo_root, args.model) if args.model is not None else newest_model(repo_root / "models").run_dir
        built = build_package(
            repo_root,
            model_dir,
            _resolve(repo_root, args.output),
            uv=find_uv(os.environ),
            allow_uncommitted=args.allow_uncommitted,
            now=datetime.now(timezone.utc),
        )
    except (BuildError, ModelError) as error:
        print(f"Cannot build the package: {error}", file=sys.stderr)
        return 1
    print(f"Package: {built.zip_path} ({built.zip_path.stat().st_size / 1e6:,.0f} MB)")
    print(
        f"Unzipped: {built.folder}, {built.files:,} files, {built.size_bytes / 1e9:.2f} GB, "
        f"longest path {built.longest_path} characters"
    )
    return 0


def _resolve(repo_root: Path, path: Path) -> Path:
    return path if path.is_absolute() else repo_root / path


if __name__ == "__main__":
    raise SystemExit(main())
